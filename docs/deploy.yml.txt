name: Deploy to GitHub Pages

# Build + publish. The widget itself has NO scraper and NO data cron — it fetches the MLB Stats
# API directly in the browser. The scheduled runs below exist only to regenerate the EMAIL
# image (public digest.png): email can't run the live widget, so it's snapshotted to a PNG on a
# schedule. This bakes an image, not a data cache — see scripts/render-digest.mjs.

on:
  push:
    branches: [main]
  workflow_dispatch:
  schedule:
    # Refresh the email image before the ~8 AM and ~5 PM Central sends, plus two overnight
    # passes so late West Coast games (final near midnight Central — every NLDS/NLCS road night
    # in San Diego or LA) reach the image before morning. Cron is UTC and does not follow DST;
    # during the season (CDT) these land ~1:15 AM / ~3:45 AM / ~6:45 AM / ~3:45 PM Central.
    # Odd minutes on purpose: GitHub delays (and under load drops) scheduled runs at the top and
    # half of the hour — in Oct 2026 the old :00/:30 ticks were firing 3–6 hours late, so the
    # "1 AM" bake was landing after breakfast. Two overnight ticks give the morning send a spare.
    - cron: '17 6 * * *'
    - cron: '47 8 * * *'
    - cron: '43 11 * * *'
    - cron: '43 20 * * *'

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
      - run: npm ci
      - run: npm run build
      # Render the digest widget to dist/digest.png for the email newsletter (see header note).
      - run: npx playwright install --with-deps chromium
      - name: Render digest email image
        id: digest
        continue-on-error: true
        run: |
          npm run preview -- --port 4173 --host 127.0.0.1 &
          for i in $(seq 1 30); do
            curl -sf http://127.0.0.1:4173/wpr-brewers-tracker/mini-digest.html > /dev/null && break
            sleep 1
          done
          node scripts/render-digest.mjs http://127.0.0.1:4173/wpr-brewers-tracker/mini-digest.html dist/digest.png
      # A failed render (MLB API unreachable from CI → the script refuses to bake a degraded card)
      # must neither block the deploy nor blank the newsletter image — carry the last published
      # digest forward instead.
      - name: Keep last good digest on render failure
        if: steps.digest.outcome == 'failure'
        run: |
          curl -fsS -o dist/digest.png https://rowanflynnpilot.github.io/wpr-brewers-tracker/digest.png || true
      - uses: actions/configure-pages@v5
      - uses: actions/upload-pages-artifact@v3
        with:
          path: dist

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      # NOTE: deploy-pages hard-caps its queue wait at 10 minutes (larger `timeout` values are
      # clamped with a warning). If the Pages queue stalls (seen 2026-07-02), the only fix is
      # re-running the deploy job until GitHub's queue clears — the twice-daily schedule also
      # self-heals it on the next tick.
      - id: deployment
        uses: actions/deploy-pages@v4
